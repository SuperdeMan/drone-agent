"""Download the pinned source archive with resumable verified ranges.

用可续传并校验的分段下载固定的源归档。
"""
import concurrent.futures
import hashlib
import json
import threading
import time
import urllib.request
from pathlib import Path

ROOT = Path(__file__).resolve().parent
META = json.loads((ROOT / 'sources/uavvaste-zenodo.json').read_text())['files'][0]
SIZE = META['size']
URL = 'https://zenodo.org/records/8214061/files/UAVVasteDataset.zip?download=1'
TARGET = ROOT / 'raw/UAVVasteDataset.zip.partial'
STATE = ROOT / 'raw/download-state.json'
LOCK = threading.Lock()
CHUNK = 32 * 1024 * 1024


def digest_file(path):
    """Hash all source bytes. / 对全部源字节计算摘要。"""
    hashes = {n: hashlib.new(n) for n in ('md5', 'sha256')}
    with path.open('rb') as f:
        for block in iter(lambda: f.read(4 * 1024 * 1024), b''):
            for h in hashes.values():
                h.update(block)
    return {n: h.hexdigest() for n, h in hashes.items()}


def main():
    """Keep a receipt; never extract an unchecked archive. / 保留回执；不解压未校验归档。"""
    state = json.loads(STATE.read_text()) if STATE.exists() else {'url': URL, 'size': SIZE, 'done': {}}
    if TARGET.exists() and TARGET.stat().st_size and not STATE.exists():
        prefix = TARGET.stat().st_size
        state['prefix'] = prefix
    start = state.get('prefix', 0)
    TARGET.touch(exist_ok=True)
    jobs = [(s, min(s + CHUNK, SIZE)) for s in range(start, SIZE, CHUNK)]
    with TARGET.open('rb') as f:
        for s, e in jobs:
            if str(s) in state['done']:
                f.seek(s)
                if hashlib.sha256(f.read(e-s)).hexdigest() != state['done'][str(s)]:
                    raise ValueError(f'previous chunk changed: {s}')
    STATE.write_text(json.dumps(state, indent=2))

    def fetch(pair):
        s, e = pair
        if str(s) in state['done']:
            return
        for attempt in range(5):
            try:
                req = urllib.request.Request(URL, headers={'Range': f'bytes={s}-{e-1}',
                                                           'User-Agent': 'drone-agent-material-audit/1'})
                with urllib.request.urlopen(req, timeout=60) as response, TARGET.open('r+b') as out:
                    expected = f'bytes {s}-{e-1}/{SIZE}'
                    if response.status != 206 or response.headers.get('Content-Range') != expected:
                        raise ValueError('server did not honor requested range')
                    out.seek(s)
                    count = 0
                    digest = hashlib.sha256()
                    while data := response.read(1024 * 1024):
                        if count + len(data) > e - s:
                            raise ValueError('oversized response')
                        out.write(data)
                        digest.update(data)
                        count += len(data)
                    if count != e-s:
                        raise ValueError(f'short range {count}/{e-s}')
                with LOCK:
                    state['done'][str(s)] = digest.hexdigest()
                    STATE.write_text(json.dumps(state, indent=2))
                    print(json.dumps({'chunks_done': len(state['done']), 'chunks_total': len(jobs)}), flush=True)
                return
            except Exception as exc:
                print(json.dumps({'offset': s, 'attempt': attempt + 1, 'error': str(exc)}), flush=True)
                if attempt == 4:
                    raise
                time.sleep(min(3 * (attempt + 1), 15))

    with concurrent.futures.ThreadPoolExecutor(max_workers=6) as pool:
        list(pool.map(fetch, jobs))
    if TARGET.stat().st_size != SIZE:
        raise ValueError('archive size mismatch')
    hashes = digest_file(TARGET)
    if 'md5:' + hashes['md5'] != META['checksum']:
        raise ValueError('official archive MD5 mismatch')
    complete = ROOT / 'raw/UAVVasteDataset.zip'
    if complete.exists():
        raise ValueError('complete archive already exists; refusing to replace it')
    TARGET.rename(complete)
    receipt = {'status': 'verified', 'source_url': URL, 'bytes': SIZE, **hashes,
               'source_metadata_sha256': hashlib.sha256((ROOT/'sources/uavvaste-zenodo.json').read_bytes()).hexdigest()}
    (ROOT/'audit/archive-receipt.json').write_text(json.dumps(receipt, indent=2))
    print(json.dumps(receipt), flush=True)


if __name__ == '__main__':
    main()
