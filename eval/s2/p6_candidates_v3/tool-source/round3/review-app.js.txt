/* Local-only annotation workbench; no fetch, upload or model call. / 仅本地标注，无请求、上传或模型调用。 */
(function(){
 'use strict';
 const data=window.REVIEW_DATA,core=window.ReviewCore,$=id=>document.getElementById(id);
 const key='p6-review:'+data.bundle_id; let index=0,mode=null,points=[],targetVisible=true,sourceVisible=false;
 let records=[],drafts=Object.create(null),reviewer='',lastTick=performance.now(),lastActivity=lastTick;
 try{const saved=JSON.parse(localStorage.getItem(key)||'null');if(saved){records=core.parseImport(JSON.stringify({format:'drone.p6-operator-reviews/v1',bundle_id:data.bundle_id,records:saved.records||[]}),data);drafts=saved.drafts||Object.create(null);reviewer=saved.reviewer||'';}}catch(e){$('storageStatus').textContent='未恢复本机缓存，请使用已导出的 JSON 文件';}
 for(const d of Object.values(drafts))if(d&&typeof d==='object')d.running=false;
 const item=()=>data.cases[index];
 function draft(){const c=item();if(!drafts[c.case_id]){const old=records.find(r=>r.case_id===c.case_id);drafts[c.case_id]=old?{...old,running:false,elapsed:old.active_review_ms}:{verdict:'',notes:'',target_region:c.target_region?{kind:'source_support',mask_sha256:c.target_region.sha256}:null,obstruction_region:null,target_confirmed:false,capture_group:c.capture_group_proposal,group_confirmed:false,source_hints_viewed:false,ai_hints_viewed:false,started_at:null,elapsed:0,running:false};}return drafts[c.case_id];}
 function persist(){try{localStorage.setItem(key,JSON.stringify({records,drafts,reviewer}));$('storageStatus').textContent='草稿已保存在本机 · 正式留存请导出 JSON';}catch(e){$('storageStatus').textContent='本机缓存不可用，请及时导出已登记记录';}}
 function notify(text){$('message').textContent=text;}
 function activity(){lastActivity=performance.now();}
 ['pointerdown','pointermove','keydown'].forEach(e=>document.addEventListener(e,activity,{passive:true}));
 function tick(){const now=performance.now(),d=draft();if(d.running&&!document.hidden&&document.hasFocus()&&now-lastActivity<60000)d.elapsed+=Math.min(now-lastTick,2000);lastTick=now;const s=Math.floor(d.elapsed/1000);$('elapsed').textContent=String(Math.floor(s/60)).padStart(2,'0')+':'+String(s%60).padStart(2,'0');}
 setInterval(tick,500);document.addEventListener('visibilitychange',()=>{tick();persist();});window.addEventListener('blur',()=>{tick();persist();});
 function nav(){const list=$('caseList');list.replaceChildren();data.cases.forEach((c,i)=>{const b=document.createElement('button');b.className=i===index?'active':'';b.setAttribute('aria-current',i===index?'true':'false');const label=document.createElement('span');label.className='itemcode';const siblings=data.cases.filter(v=>v.image_id===c.image_id);label.textContent='#'+c.image_id+(siblings.length>1?' / '+String.fromCharCode(65+siblings.indexOf(c)):'');const dot=document.createElement('span');dot.className='dot'+(records.some(r=>r.case_id===c.case_id)?' done':'');b.append(label,dot);b.onclick=()=>select(i);list.append(b);});$('progress').textContent=records.length+' / '+data.cases.length+' 已登记';}
 function select(i){tick();draft().running=false;persist();index=Math.max(0,Math.min(data.cases.length-1,i));mode=null;points=[];sourceVisible=false;$('sourceImage').hidden=true;$('maskToggle').setAttribute('aria-pressed','false');$('aiPanel').open=false;$('zoom').value=100;zoom();render();}
 function paint(){const svg=$('geometry');svg.replaceChildren();const d=draft();function poly(p,color,open){if(!p||!p.length)return;const el=document.createElementNS('http://www.w3.org/2000/svg',open?'polyline':'polygon');el.setAttribute('points',p.map(v=>[v[0]*4000,v[1]*3000].join(',')).join(' '));el.setAttribute('fill',open?'none':color+'24');el.setAttribute('stroke',color);el.setAttribute('stroke-width','9');svg.append(el);p.forEach(v=>{const dot=document.createElementNS('http://www.w3.org/2000/svg','circle');dot.setAttribute('cx',v[0]*4000);dot.setAttribute('cy',v[1]*3000);dot.setAttribute('r','15');dot.setAttribute('fill',color);svg.append(dot);});}
   if(targetVisible && d.target_region?.kind==='polygon')poly(d.target_region.points,'#319fdc',false);
   poly(d.obstruction_region?.points,'#f79536',false);poly(points,mode==='obstacle'?'#f79536':'#319fdc',true);
   $('targetImage').hidden=!(targetVisible&&d.target_region?.kind==='source_support'&&item().target_preview);
   $('drawTarget').setAttribute('aria-pressed',mode==='target'?'true':'false');$('drawObstacle').setAttribute('aria-pressed',mode==='obstacle'?'true':'false');
   $('geometryStatus').textContent=mode?('正在描绘'+(mode==='target'?'目标':'障碍')+' · '+points.length+' 个顶点，完成后点击“完成轮廓”'):d.target_region?.kind==='polygon'?'使用自定义目标区域':d.target_region?'使用源道路可见支持域':'尚无目标区域';
 }
 function render(){const c=item(),d=draft();nav();$('caseTitle').textContent='影像 #'+c.image_id;$('caseMeta').textContent=c.capture_group_proposal+' · 4000 × 3000 · 开发素材';$('originalLink').href=c.original;$('photo').src=c.preview;$('sourceImage').src=c.source_overlay;if(c.target_preview)$('targetImage').src=c.target_preview;else $('targetImage').removeAttribute('src');$('emptyHint').hidden=!!d.target_region;
   $('reviewer').value=reviewer;$('verdict').value=d.verdict;$('notes').value=d.notes;$('group').value=d.capture_group;$('targetConfirmed').checked=d.target_confirmed;$('groupConfirmed').checked=d.group_confirmed;$('startTimer').textContent=d.running?'暂停计时':d.started_at?'继续复核':'开始复核';
   $('aiText').textContent=({normal_candidate:'开发建议：未见明确阻塞候选。',suspected_candidate:'开发建议：疑似障碍候选。',unknown_candidate:'开发建议：证据不足，建议保留不可判。',wrong_target_candidate:'开发建议：错目标候选。'}[c.ai_suggestion]||'')+' '+c.ai_notes;
   $('license').textContent=data.license_note;$('evidence').textContent='case: '+c.case_id+'\nimage sha256: '+c.image_sha256;$('position').textContent=(index+1)+' / '+data.cases.length;$('previous').disabled=index===0;$('next').disabled=index===data.cases.length-1;notify(records.some(r=>r.case_id===c.case_id)?'本条已有登记；修改后需再次登记。':'尚未登记复核结论。');paint();tick();
 }
 function zoom(){const percent=Number($('zoom').value);$('stage').style.width=percent+'%';$('zoomValue').textContent=percent+'%';}
 $('zoom').oninput=zoom;$('previous').onclick=()=>select(index-1);$('next').onclick=()=>select(index+1);
 $('photo').onerror=()=>{$('imageError').style.display='block';};$('photo').onload=()=>{$('imageError').style.display='none';};
 $('reviewer').oninput=()=>{reviewer=$('reviewer').value;persist();};
 [['verdict','verdict'],['notes','notes'],['group','capture_group']].forEach(([id,field])=>{$(id).oninput=()=>{draft()[field]=$(id).value;persist();};});
 [['targetConfirmed','target_confirmed'],['groupConfirmed','group_confirmed']].forEach(([id,field])=>{$(id).onchange=()=>{draft()[field]=$(id).checked;persist();};});
 $('startTimer').onclick=()=>{tick();const d=draft();d.running=!d.running;if(d.running&&!d.started_at)d.started_at=new Date().toISOString();activity();$('startTimer').textContent=d.running?'暂停计时':'继续复核';persist();notify(d.running?'计时开始；切换标签或 60 秒无交互会暂停累计。':'计时已暂停。');};
 $('targetToggle').onclick=()=>{targetVisible=!targetVisible;$('targetToggle').setAttribute('aria-pressed',String(targetVisible));paint();};
 $('maskToggle').onclick=()=>{sourceVisible=!sourceVisible;$('sourceImage').hidden=!sourceVisible;$('maskToggle').setAttribute('aria-pressed',String(sourceVisible));if(sourceVisible){draft().source_hints_viewed=true;persist();notify('源标签已查看：绿色为畅通道路，红色为阻塞道路；不是障碍物轮廓。');}};
 $('aiPanel').ontoggle=()=>{if($('aiPanel').open){draft().ai_hints_viewed=true;persist();}};
 $('drawTarget').onclick=()=>{mode='target';points=[];paint();};$('drawObstacle').onclick=()=>{mode='obstacle';points=[];paint();};
 $('geometry').onclick=e=>{if(!mode)return;const box=$('geometry').getBoundingClientRect();points.push([Math.max(0,Math.min(1,(e.clientX-box.left)/box.width)),Math.max(0,Math.min(1,(e.clientY-box.top)/box.height))]);paint();};
 $('undoPoint').onclick=()=>{points.pop();paint();};
 $('finishDraw').onclick=()=>{if(!mode){notify('先选择描绘目标或圈出障碍。');return;}if(!core.polygonValid(points)){notify('至少需要三个不共线、不自交的顶点。');return;}const geometry={kind:'polygon',points:points.map(p=>p.map(v=>Math.round(v*1000000)/1000000))};if(mode==='target'){draft().target_region=geometry;draft().target_confirmed=false;$('targetConfirmed').checked=false;}else draft().obstruction_region=geometry;mode=null;points=[];persist();paint();$('emptyHint').hidden=!!draft().target_region;notify('轮廓已保存为本机草稿，尚未登记结论。');};
 $('clearDrawing').onclick=()=>{const c=item(),d=draft();d.target_region=c.target_region?{kind:'source_support',mask_sha256:c.target_region.sha256}:null;d.obstruction_region=null;d.target_confirmed=false;$('targetConfirmed').checked=false;mode=null;points=[];persist();paint();notify('已恢复本条源支持域，清除自绘轮廓；已有登记记录未改变。');};
 $('saveReview').onclick=()=>{tick();const c=item(),d=draft();const record={case_id:c.case_id,image_sha256:c.image_sha256,reviewer:reviewer.trim(),verdict:d.verdict,notes:d.notes.trim(),target_region:d.target_region,obstruction_region:d.obstruction_region,target_confirmed:!!d.target_confirmed,capture_group:d.capture_group,group_confirmed:!!d.group_confirmed,source_hints_viewed:!!d.source_hints_viewed,ai_hints_viewed:!!d.ai_hints_viewed,started_at:d.started_at,completed_at:new Date().toISOString(),active_review_ms:Math.round(d.elapsed),review_origin:'operator_entered',revision:(records.find(r=>r.case_id===c.case_id)?.revision||0)+1};
   const errors=core.validate(record,c);if(errors.length){notify(errors.join('；'));return;}records=records.filter(r=>r.case_id!==c.case_id).concat(record);d.running=false;$('startTimer').textContent='继续复核';persist();nav();notify('本条已登记在本机。请导出 JSON 留存；这不表示已准出。');};
 $('exportBtn').onclick=()=>{try{const value=core.makeExport(data,records);const blob=new Blob([JSON.stringify(value,null,2)],{type:'application/json'});const url=URL.createObjectURL(blob);const a=document.createElement('a');a.href=url;a.download='p6-road-review-'+new Date().toISOString().replace(/[:.]/g,'-')+'.json';a.click();setTimeout(()=>URL.revokeObjectURL(url),2000);notify('已导出 '+records.length+' 条登记记录。未登记的草稿不计为复核完成。');}catch(e){notify(e.message);}};
 $('importBtn').onclick=()=>{$('importFile').click();};$('importFile').onchange=async()=>{try{const file=$('importFile').files[0];if(!file)return;const imported=core.parseImport(await file.text(),data);const incoming=new Map(imported.map(r=>[r.case_id,r]));records=records.filter(r=>!incoming.has(r.case_id)).concat(imported);for(const r of imported)delete drafts[r.case_id];persist();render();notify('已导入 '+imported.length+' 条；同条目按导入版本替换。');}catch(e){notify('未导入：'+e.message);}finally{$('importFile').value='';}};
 render();
})();
