def ssh(connection,code,timeout=60):
    """Do not forward the host login banner. / 不转发主机登录横幅。"""
    result=subprocess.run(['ssh',*connection.arguments(),connection.target,'python3 -'],input=code.encode(),capture_output=True,timeout=timeout)
    if result.returncode:
        raise RuntimeError(f'remote process failed with exit {result.returncode}; {result.stdout.decode(errors="replace")[-2000:]}')
    return result.stdout
