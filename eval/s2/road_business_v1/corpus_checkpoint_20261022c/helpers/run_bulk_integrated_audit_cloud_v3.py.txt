def ssh_call(code, phase, timeout):
    started = now()
    try:
        child = subprocess.run(['ssh', *connection.arguments(), connection.target, 'python3 -u -'],
            input=code.encode('utf-8'), capture_output=True, timeout=timeout)
    except subprocess.TimeoutExpired as error:
        save(CONTROL / (phase + '-ssh-timeout.json'), {'started_at': started, 'finished_at': now(),
            'timeout_seconds': timeout, 'stdout_sha256': hashlib.sha256(error.stdout or b'').hexdigest(),
            'stdout_bytes': len(error.stdout or b''), 'stderr_sha256': hashlib.sha256(error.stderr or b'').hexdigest(),
            'stderr_bytes': len(error.stderr or b''), 'raw_ssh_stderr_printed': False, 'automatic_rerun': False})
        raise RuntimeError('SSH timed out; state is unknown and no automatic audit rerun was attempted.') from None
    receipt = {'started_at': started, 'finished_at': now(), 'ssh_exit_code': child.returncode,
        'stdout_sha256': hashlib.sha256(child.stdout).hexdigest(), 'stdout_bytes': len(child.stdout),
        'stderr_sha256': hashlib.sha256(child.stderr).hexdigest(), 'stderr_bytes': len(child.stderr),
        'raw_ssh_stderr_printed': False}
    save(CONTROL / (phase + '-ssh-receipt.json'), receipt)
    if child.returncode != 0:
        raise RuntimeError('SSH phase failed; only diagnostic hashes and lengths were recorded: ' + phase)
    try:
        return json.loads(child.stdout)
    except (ValueError, UnicodeError):
        raise RuntimeError('SSH phase returned no valid JSON; diagnostics remain withheld: ' + phase) from None
