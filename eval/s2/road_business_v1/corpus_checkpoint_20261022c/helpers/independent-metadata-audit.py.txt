def sha(path):
    with path.open('rb') as handle:
        return hashlib.file_digest(handle, 'sha256').hexdigest()


def canonical(value):
    return hashlib.sha256(json.dumps(value, sort_keys=True, ensure_ascii=False,
        separators=(',', ':'), allow_nan=False).encode()).hexdigest()


def locate(value):
    text = str(value).replace('\\', '/')
    for prefix in ('D:/drone-agent-data/',
        '/home/ubuntu/drone-agent-data/p6-corpus-completion-20261018-cloud/data/'):
        if text.startswith(prefix):
            text = text[len(prefix):]
            break
    path = Path(text)
    if not path.is_absolute():
        path = DATA / path
    path = path.resolve()
    assert path.is_relative_to(ROOT) or path.is_relative_to(REPO.resolve()), str(path)
    assert path.is_file(), str(path)
    return path


def check(condition, name, sample=None, details=None):
    if not condition:
        FAILURES.append({'check': name, 'sample_id': sample, 'details': details})
    return condition


def bind(value, expected=None, kind='metadata'):
    path = locate(value)
    key = str(path)
    actual = sha(path)
    if key not in FILES:
        FILES[key] = {'file': key, 'sha256': actual, 'bytes': path.stat().st_size,
            'read_scope': 'bytes_only_no_pixel_decode' if path.suffix.lower() in
            {'.jpg', '.jpeg', '.png', '.tif', '.tiff'} else kind}
    if expected is not None:
        check(actual == expected, 'actual_file_hash_matches_declared_binding', details={
            'file': key, 'expected_sha256': expected, 'actual_sha256': actual})
    return path


def read(value, expected=None, kind='parsed_metadata'):
    return json.loads(bind(value, expected, kind).read_text(encoding='utf-8'))


def ids(rows, key='sample_id'):
    values = [row[key] for row in rows]
    check(len(values) == len(set(values)), 'rows_have_unique_ids', details=values)
    return values


def time(value):
    return datetime.fromisoformat(value.replace('Z', '+00:00'))


def close(values, expected, tolerance=1e-12):
    return len(values) == len(expected) and all(math.isclose(a, b, abs_tol=tolerance,
        rel_tol=0) for a, b in zip(values, expected))


def footprint(receipt):
    ident = receipt['sample_id']
    parent = receipt['raster']['transform']
    supplied = receipt['window_transform']
    x, y, w, h = receipt['native_window_xywh']
    check([receipt['width'], receipt['height']] == [w, h], 'native_window_dimensions', ident)
    check(x >= 0 and y >= 0 and x + w <= receipt['raster']['width']
        and y + h <= receipt['raster']['height'], 'native_window_inside_parent', ident)
    # Derive from parent affine rather than trusting supplied window affine. / 从父仿射独立复算窗口。
    derived = list(parent)
    derived[2] += x * parent[0] + y * parent[1]
    derived[5] += x * parent[3] + y * parent[4]
    check(close(derived, supplied, 1e-8), 'parent_affine_reproduces_window_affine', ident)
    a, b, c, d, e, f, _, _, _ = derived
    corners = [(a * px + b * py + c, d * px + e * py + f)
        for px, py in ((0, 0), (w, 0), (w, h), (0, h))]
    native = [min(v[0] for v in corners), min(v[1] for v in corners),
        max(v[0] for v in corners), max(v[1] for v in corners)]
    geographic = list(transform_bounds(receipt['raster']['crs'], 'EPSG:4326', *native,
        densify_pts=21))
    # Independently sample 1001 positions along each bbox edge. / 独立密集抽样四边包络。
    left, bottom, right, top = native
    edge_x, edge_y = [], []
    for index in range(1001):
        fraction = index / 1000
        px = left + fraction * (right - left)
        py = bottom + fraction * (top - bottom)
        edge_x.extend((px, px, left, right))
        edge_y.extend((bottom, top, py, py))
    lon, lat = transform(receipt['raster']['crs'], 'EPSG:4326', edge_x, edge_y)
    dense = [min(lon), min(lat), max(lon), max(lat)]
    check(geographic[0] <= dense[0] + 1e-12 and geographic[1] <= dense[1] + 1e-12
        and geographic[2] >= dense[2] - 1e-12 and geographic[3] >= dense[3] - 1e-12,
        '21_point_envelope_covers_1001_point_edges', ident)
    return {'sample_id': ident, 'source_crs': receipt['raster']['crs'],
        'native_image_bounds': native, 'conservative_full_image_envelope_lon_lat': geographic,
        'independent_1001_per_edge_bounds_lon_lat': dense,
        'full_native_derived_image_used_not_parent_tiff': True}


def declared_footprint_matches(declared, actual):
    return (declared['sample_id'] == actual['sample_id']
        and declared['source_crs'] == actual['source_crs']
        and close(declared['native_image_bounds'], actual['native_image_bounds'], 1e-8)
        and close(declared['conservative_full_image_envelope_lon_lat'],
            actual['conservative_full_image_envelope_lon_lat']))


def source_binding(receipt, slug, source_context):
    ident = receipt['sample_id']
    binding = read(ROOT / f'resume-20261021/{slug}-bindings-v1/{ident}-binding.json')
    for name in ('extraction', 'extraction_plan', 'source_inventory', 'range_manifest', 'replay'):
        bind(binding[name + '_file'], binding[name + '_sha256'])
    check(binding['extraction_sha256'] == sha(locate(source_context['extraction_file'])),
        'prior_source_binding_is_current_extraction', ident)
    context = source_context['source']
    prior = context['capture_rows'][ident]
    check(canonical(binding) == prior['derivation_sha256'], 'f132_inventory_binds_exact_derivation', ident)
    for key, actual in (
        ('image_sha256', receipt['image_sha256']), ('source_orthomosaic', receipt['source_orthomosaic']),
        ('source_commit', receipt['source_commit']), ('parent_sha256', receipt['source_published_lfs_sha256'])):
        check(prior[key] == actual, 'f132_capture_inventory_matches_current_' + key, ident)
    check(ident in context['passed_ids'], 'current_id_in_actual_prior_f132_source_pass', ident)
    check(prior['observed']['asset_id'] == receipt['asset_group_id'], 'f132_inventory_preserves_group', ident)
    plan = read(binding['extraction_plan_file'])
    source_plan = read(plan['source_plan_file'], plan['source_plan_sha256'])
    planned = [item for source in source_plan['images'] for item in source['items']
        if item['bounded_target_id'] == receipt['bounded_target_id']
        and item['asset_group_id'] == receipt['asset_group_id']]
    check(len(planned) == 1 and planned[0]['target'] == receipt['world_scope'],
        'current_scope_matches_preregistered_extraction_plan', ident)
    replay = read(binding['replay_file'])
    replayed = read(replay['replay_extraction_file'], replay['replay_extraction_sha256'])
    repeated_keys = ('sample_id', 'asset_group_id', 'bounded_target_id', 'source_orthomosaic',
        'source_commit', 'source_published_lfs_sha256', 'media_origin', 'status', 'world_scope',
        'raster', 'native_window_xywh', 'window_transform', 'width', 'height', 'image_sha256',
        'mask_sha256', 'native_rgb_sha256', 'valid_pixel_fraction', 'source_geometry_sha256',
        'source_alignment_sha256', 'source_date', 'source_date_double_check_flag')
    check(all(receipt[k] == replayed[k] for k in repeated_keys),
        'prior_offline_replay_receipt_matches_current_extraction', ident)
    check(replay['network_requests'] == 0 and replay['mode'] == 'offline_cached_ranges',
        'saved_replay_declares_zero_network', ident)
    for name in ('image', 'mask', 'guide', 'overview', 'source_alignment', 'source_geometry'):
        bind(receipt[name + '_file'], receipt[name + '_sha256'], kind='bytes_only')
    ranges = read(binding['range_manifest_file'])
    return {'sample_id': ident, 'source_slug': slug,
        'source_report_file': str(context['report_file']), 'source_report_sha256': sha(context['report_file']),
        'current_derivation_canonical_sha256': canonical(binding),
        'capture_inventory_file': str(context['capture_file']),
        'capture_inventory_sha256': sha(context['capture_file']),
        'source_chain_was_verified_by_prior_f132_run': True,
        'prior_capture_author_date_group': prior['author_date_group'],
        'prior_capture_author_date_verified': prior['author_date_verified'],
        'current_audit_decoded_pixels': False,
        'whole_parent_sha256_verified': ranges.get('whole_parent_sha256_verified'),
        'new_repeat_evidence_created': False}

            observer_rows = {v['sample_id']: v for v in payload['rows']}
