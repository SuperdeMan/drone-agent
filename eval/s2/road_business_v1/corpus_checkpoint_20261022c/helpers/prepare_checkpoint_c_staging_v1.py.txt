def sha(path):
    with path.open('rb') as handle:
        return hashlib.file_digest(handle, 'sha256').hexdigest()


def read(path):
    return json.loads(path.read_text(encoding='utf-8'))


def canonical(value):
    return hashlib.sha256(json.dumps(value, sort_keys=True, ensure_ascii=False,
        separators=(',', ':'), allow_nan=False).encode()).hexdigest()


def source_path(value):
    text = str(value).replace('\\', '/')
    for prefix in ('D:/drone-agent-data/',
        '/home/ubuntu/drone-agent-data/p6-corpus-completion-20261018-cloud/data/'):
        if text.startswith(prefix):
            text = text[len(prefix):]
            break
    path = Path(text)
    return (path if path.is_absolute() else DATA / path).resolve()


def reference(path, purpose, link=None):
    assert path.is_file()
    return {'source_file': str(path), 'sha256': sha(path), 'bytes': path.stat().st_size,
        'purpose': purpose, 'archive_relative_link': link}

def add(name, path):
    assert name not in copies and name not in generated
    assert path.is_file(), str(path)
    assert path.suffix.lower() not in {'.png','.jpg','.jpeg','.tif','.tiff','.gz','.zip'}

    generated[name] = (json.dumps(value, ensure_ascii=False, indent=2, allow_nan=False) + '\n').encode('utf-8')
