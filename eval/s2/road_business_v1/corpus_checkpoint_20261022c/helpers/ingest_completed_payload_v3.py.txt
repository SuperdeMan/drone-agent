"""Store explicitly completed observer payloads without making visual judgments.

存储观察者明确完成的批次，不新增视觉判断。
"""
import argparse
import copy
import hashlib
import json
import runpy
import subprocess
import sys
from datetime import UTC,datetime
from pathlib import Path

ROOT=Path(__file__).resolve().parent.parent
DATA=ROOT.parent
REPO=Path('D:/drone-agent-data/p6-road-views-20261014/repo')

def write_chunks(command, values, destination):
    """Persist actual rows in small batches without changing business fields. / 原样分批保存。"""
    for start in range(0,len(values),10):
        subset=values[start:start+10]
        path=destination/('completed-observer-chunk-%04d-%04d.json'%(start+1,start+len(subset)))
        with path.open('x',encoding='utf-8',newline='\n') as handle:json.dump(subset,handle,ensure_ascii=False,indent=2)
        actual=list(command)
        actual[actual.index('--batch')+1]=str(path)
        actual[actual.index('--expected-count')+1]=str(start)
        subprocess.run(actual,cwd=ROOT,check=True)

read=lambda path:json.loads(path.read_text(encoding='utf-8'))
sha=lambda path:hashlib.sha256(path.read_bytes()).hexdigest()
C=runpy.run_path(str(REPO/'scripts/p6_business_eval.py'))
parser=argparse.ArgumentParser()
parser.add_argument('--role',choices=['first','review'],required=True)
parser.add_argument('--worker',required=True)
args=parser.parse_args()
folder=ROOT/('annotation' if args.role=='first' else 'verification')/args.worker
assignment_file=folder/'assignment.json';assignment=read(assignment_file)
payload_file=folder/('first-payload.json' if args.role=='first' else 'review-payload.json')
payload=read(payload_file)
ident_key='author_id' if args.role=='first' else 'reviewer_id'
assert payload[ident_key]==args.worker and payload['assignment_sha256']==sha(assignment_file)
assert payload['status']=='complete' and payload['human_reviews_created']==0 and payload['formal_release_eligible'] is False
assert payload['full_assigned_targets_checked' if args.role=='first' else 'full_assigned_target_checked'] is True
rows=copy.deepcopy(payload['rows']);assigned={row['sample_id']:row for row in assignment['samples']}
assert {row['sample_id'] for row in rows}==set(assigned) and len(rows)==len(assigned)
provenance={'operator':'root_orchestrator_deterministic_storage_only','observer_declared_status':'complete',
    'observer_payload_file':str(payload_file),'observer_payload_sha256':sha(payload_file),
    'payload_file_modified_at_utc':datetime.fromtimestamp(payload_file.stat().st_mtime,UTC).isoformat(),
    'filesystem_time_is_not_claimed_visual_inspection_time':True,'stored_at_utc':datetime.now(UTC).isoformat(),
    'business_fields_changed':False,'bounded_writer_chunks_maximum':10,'base_ingestion_source_sha256':'d4a346bac670e39e8157827f10aaaa1d24f22fbfe5db6e085d52ed9db2c4867c','new_visual_inspection_claimed':False,'observer_storage_cli_execution_claimed':False}
for row in rows:
    source=assigned[row['sample_id']]
    assert sha(Path(source['image_file']))==source['image_sha256']
    files=row['inspection_files'] if args.role=='first' else row['inspected_files']
    resolved={Path(value).resolve() for value in files}; assert Path(source['image_file']).resolve() in resolved and (not source.get('guide_file') or Path(source['guide_file']).resolve() in resolved)
    if args.role=='first' and source.get('expected_reference_image_file'):assert Path(source['expected_reference_image_file']).resolve() in resolved
    for value in files:
        path=Path(value).resolve();assert path.is_relative_to(DATA.resolve()) and path.is_file()
        assert path.suffix.lower() not in {'.env','.pem','.key'}

# Reject malformed fields before creating persisted observer captures. / 保存前拒绝结构错误。
for item in rows:
    final=item if args.role=='first' else {**assigned[item['sample_id']]['first_row'],**item['changes']}
    for field in ('uncertainties','inspection_files'):
        assert isinstance(final[field],list) and all(isinstance(value,str) and value.strip() for value in final[field])
    for field in ('target_brief','notes','uncertainties'):
        texts=final[field] if field=='uncertainties' else [final[field]]
        assert all('\ufffd' not in value and '???' not in value and '\uff1f\uff1f\uff1f' not in value for value in texts)
    C['ai_suggestions']({'format':'drone.p6-road-ai-suggestions/v1','author_kind':'isolated_ai_annotator','author_id':args.worker,'created_at':datetime.now(UTC).isoformat(),'human_reviews_created':0,'formal_release_eligible':False,'isolation':{'parent_history':'none','evaluation_prompt_seen':False,'evaluation_outputs_seen':False,'source_labels_seen':False},'rows':[final]})

captured=folder/'completed-observer-payload-captured-v1.json'
with captured.open('xb') as handle:handle.write(payload_file.read_bytes())
assert sha(captured)==sha(payload_file)
raw_batch=folder/'completed-observer-rows-v1.json'
with raw_batch.open('x',encoding='utf-8',newline='\n') as handle:json.dump(rows,handle,ensure_ascii=False,indent=2)
if args.role=='first':
    provenance['inspection_path_spelling_normalized']=True
    provenance['inspection_path_rule']='Resolve equivalent path separators only; original payload bytes are retained and every resolved file was checked within the data root.'
    for row in rows:row['inspection_files']=[str(Path(value).resolve()) for value in row['inspection_files']]
    with raw_batch.with_name('completed-observer-rows-canonical-paths-v1.json').open('x',encoding='utf-8',newline='\n') as handle:json.dump(rows,handle,ensure_ascii=False,indent=2)
    raw_batch=raw_batch.with_name('completed-observer-rows-canonical-paths-v1.json')
    assert not (folder/'suggestions.json').exists()
    for row in rows:
        assert C['timestamp'](row['inspected_at'])>=C['timestamp'](assignment['created_at'])
        source=assigned[row['sample_id']]
        if source.get('target_selection')=='pre_registered_prior_task':
            assert all(row[key]==source[key] for key in ['target_brief','target_polygon'])
    command=[sys.executable,str(ROOT/'resume-20261021/strict_batch_io_v1.py'),'annotation','--assignment',str(assignment_file),
        '--batch',str(raw_batch),'--expected-count','0']
    write_chunks(command,rows,folder)
    output=folder/'suggestions.json';document=read(output)
    assert document['rows']==rows
    document['persistence_provenance']=provenance
else:
    assert not (folder/'reviews.json').exists()
    command=[sys.executable,str(ROOT/'resume-20261021/strict_batch_io_v1.py'),'verification','append','--assignment',str(assignment_file),
        '--batch',str(raw_batch),'--expected-count','0']
    write_chunks(command,rows,folder)
    output=folder/'reviews.json';document=read(output)
    by_id={row['sample_id']:row for row in rows}
    for record in document['rows']:
        raw=by_id[record['sample_id']];source=assigned[record['sample_id']]
        expected={**source['first_row'],**raw['changes']}
        for key in ['sample_id','image_sha256','target_brief','target_polygon','verdict','objects','notes','uncertainties']:
            assert record['final_row'][key]==expected[key],key
        assert record['findings']==raw['findings'] and record['changes']==raw['changes']
        record['storage_provenance']={**provenance,'timing_semantics':'reviewed_at/final.inspected_at are storage timestamps, not an additional visual inspection or human timing.'}
    document['persistence_events']=[provenance]
with output.open('w',encoding='utf-8',newline='\n') as handle:json.dump(document,handle,ensure_ascii=False,indent=2)
receipt={'recorded_at':datetime.now(UTC).isoformat(),'role':args.role,'worker_id':args.worker,
    'source_payload_sha256':sha(payload_file),'captured_payload_sha256':sha(captured),
    'output_file':str(output),'output_sha256':sha(output),'rows':len(document['rows']),
    'business_fields_unchanged':True,'new_visual_inspection_claimed':False,'human_reviews_created':0,'formal_release_eligible':False}
with (folder/'completed-payload-ingestion-receipt-v1.json').open('x',encoding='utf-8',newline='\n') as handle:json.dump(receipt,handle,indent=2)
print(json.dumps(receipt))
