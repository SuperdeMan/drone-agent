"""Propose repeated-view comparisons from metadata, without declaring pairs verified.

用元数据提出重复视角比较，不自动确认跨次配对或资产身份。
"""
import hashlib
import json
import math
from datetime import UTC, datetime
from pathlib import Path

ROOT = Path(__file__).resolve().parent
DATA = Path('D:/drone-agent-data')

def read(path):
    return json.loads(path.read_text(encoding='utf-8-sig'))

def distance(a, b):
    x1, y1 = map(math.radians, a)
    x2, y2 = map(math.radians, b)
    return 6371000 * 2 * math.asin(min(1, math.sqrt(math.sin((x2-x1)/2)**2 + math.cos(x1)*math.cos(x2)*math.sin((y2-y1)/2)**2)))

registration = read(ROOT / 'registration-v2.json')
reviews = {}
for selected in registration['selection']:
    if selected['reviewer_id'] == 'independent_road_verifier_4':
        continue
    for row in read(Path(selected['file']))['rows']:
        reviews[row['sample_id']] = row
inventory_path = DATA / 'p6-business-closeout-20261015/inventory-v4.json'
inventory = {row['image_id']: row for row in read(inventory_path)['rows']}
eligible = [ident for ident, row in reviews.items() if row['final_row']['verdict'] in {'normal', 'obstacle'}
            and inventory[ident.split('-')[-1]].get('gps') and inventory[ident.split('-')[-1]].get('dhash256')]
candidates = []
for n, left in enumerate(eligible):
    first = inventory[left.split('-')[-1]]
    for right in eligible[n+1:]:
        second = inventory[right.split('-')[-1]]
        meters = distance(first['gps'], second['gps'])
        if meters > 100 or first['capture_time_text'][:10] != second['capture_time_text'][:10]:
            continue
        before, after = reviews[left]['final_row'], reviews[right]['final_row']
        if before['verdict'] != after['verdict'] or {o['defect_type'] for o in before['objects']} != {o['defect_type'] for o in after['objects']}:
            continue
        bits = (int(first['dhash256'], 16) ^ int(second['dhash256'], 16)).bit_count()
        if bits <= 95:
            candidates.append((bits, meters, left, right))
candidates.sort()
selected, uses = [], {}
for bits, meters, left, right in candidates:
    if max(uses.get(left, 0), uses.get(right, 0)) >= 2:
        continue
    selected.append({'pair_id': f'repeat-candidate-{len(selected)+1:03d}', 'sample_ids': [left, right],
                     'gps_distance_m': round(meters, 2), 'dhash256_distance': bits})
    for ident in (left, right):
        uses[ident] = uses.get(ident, 0) + 1
    if len(selected) == 30:
        break
images = {}
for ident in uses:
    source, review = inventory[ident.split('-')[-1]], reviews[ident]
    images[ident] = {'sample_id': ident, 'image_file': str(DATA / source['file']), 'image_sha256': source['image_sha256'],
        'capture_time_text': source['capture_time_text'], 'capture_timezone': source.get('capture_timezone'),
        'source_observation_id': source['image_id'], 'gps': source['gps'], 'reviewed_reference': review['final_row']}
value = {'format': 'drone.p6-repeat-candidate-assignment/v1', 'reviewer_id': 'independent_repeat_verifier_1',
    'created_at': datetime.now(UTC).isoformat(), 'parent_history': 'none',
    'evaluation_prompt_access': 'forbidden', 'evaluation_outputs_access': 'forbidden', 'source_class_label_access': 'forbidden',
    'source_inventory_file': str(inventory_path), 'source_inventory_sha256': hashlib.sha256(inventory_path.read_bytes()).hexdigest(),
    'selection': 'metadata-only nearest perceptual candidates <=100m with matching reviewed deterministic state/types; at most two appearances per original; not identity verification',
    'pairs': selected, 'images': images, 'human_reviews_created': 0, 'formal_release_eligible': False}
with (ROOT / 'repeat-candidates.json').open('x', encoding='utf-8') as stream:
    json.dump(value, stream, ensure_ascii=False, indent=2)
print(json.dumps({'pairs': len(selected), 'originals': len(images), 'candidate_pool': len(candidates)}))
