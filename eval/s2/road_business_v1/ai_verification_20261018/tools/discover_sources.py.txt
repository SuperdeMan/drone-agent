"""Archive public source metadata without running remote dataset code.

归档公开来源元数据，不执行远端数据集代码。
"""
import hashlib
import json
import urllib.request
from concurrent.futures import ThreadPoolExecutor
from datetime import UTC, datetime
from pathlib import Path

ROOT = Path(__file__).resolve().parent / 'source-discovery'
ROOT.mkdir(exist_ok=True)
URLS = {
    'ladi-s3-root.xml': 'https://ladi.s3.amazonaws.com/?list-type=2&delimiter=/',
    'ladi-author-readme.md': 'https://raw.githubusercontent.com/LADI-Dataset/ladi-overview/main/README.md',
    'ladi-aws-registry.html': 'https://registry.opendata.aws/ladi/',
    'ladi-hf-script-tree.json': 'https://huggingface.co/api/datasets/MITLL/LADI-v2-dataset/tree/script',
    'crasar-api.json': 'https://huggingface.co/api/datasets/CRASAR/CRASAR-U-DROIDs',
    'crasar-road-readme.md': 'https://huggingface.co/datasets/CRASAR/CRASAR-U-DROIDs/raw/main/README.md',
}

def fetch(item):
    name, url = item
    result = {'file': name, 'url': url, 'started_at': datetime.now(UTC).isoformat()}
    try:
        request = urllib.request.Request(url, headers={'User-Agent': 'drone-agent-public-data-review/1.0'})
        with urllib.request.urlopen(request, timeout=30) as response:
            data = response.read(8 * 1024 * 1024 + 1)
        if len(data) > 8 * 1024 * 1024:
            raise ValueError('metadata exceeds bounded discovery size')
        with (ROOT / name).open('xb') as stream:
            stream.write(data)
        result.update(status='downloaded', bytes=len(data), sha256=hashlib.sha256(data).hexdigest())
    except Exception as error:
        result.update(status='failed', error=f'{type(error).__name__}: {error}')
    result['finished_at'] = datetime.now(UTC).isoformat()
    return result

with ThreadPoolExecutor(max_workers=4) as pool:
    results = list(pool.map(fetch, URLS.items()))
with (ROOT / 'receipt.json').open('x', encoding='utf-8') as stream:
    json.dump({'sources': results, 'images_acquired': 0, 'formal_release_eligible': False}, stream, indent=2)
print(json.dumps(results))
