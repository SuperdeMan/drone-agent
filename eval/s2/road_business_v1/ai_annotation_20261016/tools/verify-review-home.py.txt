"""Check the navigation page without opening any original photographs.

核对导航页面，不打开任何真实照片。
"""
import asyncio
import hashlib
import json
from pathlib import Path

from playwright.async_api import async_playwright


async def main():
    root = Path(__file__).resolve().parent
    output = root / 'home-qa'
    output.mkdir(exist_ok=False)
    errors, requests, checks = [], [], []
    async with async_playwright() as pw:
        browser = await pw.chromium.launch(channel='msedge', headless=True)
        context = await browser.new_context(viewport={'width': 1280, 'height': 900})
        page = await context.new_page()
        page.on('pageerror', lambda e: errors.append(str(e)))
        page.on('request', lambda r: requests.append(r.url))
        await page.goto((root / 'index.html').as_uri(), wait_until='load')
        links = await page.locator('a.batch').evaluate_all('(nodes)=>nodes.map(n=>n.getAttribute("href"))')
        checks.append({'name': 'three_existing_review_links', 'passed': links == [f'review-batch{n}/index.html' for n in (1, 2, 3)] and all((root / p).is_file() for p in links)})
        for width, height in [(1280, 900), (390, 844)]:
            await page.set_viewport_size({'width': width, 'height': height})
            await page.screenshot(path=str(output / f'home-{width}.png'), full_page=True)
            checks.append({'name': f'no_horizontal_overflow_{width}', 'passed': await page.evaluate('document.documentElement.scrollWidth<=innerWidth+1')})
        await browser.close()
    checks += [{'name': 'no_javascript_errors', 'passed': not errors},
               {'name': 'navigation_only_no_real_photographs', 'passed': requests == [(root / 'index.html').as_uri()]}]
    receipt = {'scope': 'static local navigation only; no real photos opened', 'checks': checks, 'errors': errors,
               'passed': sum(c['passed'] for c in checks), 'failed': sum(not c['passed'] for c in checks),
               'page_sha256': hashlib.sha256((root / 'index.html').read_bytes()).hexdigest(),
               'human_reviews_created': 0, 'formal_release_eligible': False}
    (output / 'receipt.json').write_text(json.dumps(receipt, indent=2) + '\n', encoding='utf-8', newline='\n')
    print(json.dumps(receipt))
    if receipt['failed']:
        raise SystemExit(1)


asyncio.run(main())
