now = lambda: datetime.now(timezone.utc).isoformat()


def sha(path):
    with pathlib.Path(path).open('rb') as handle:
        return hashlib.file_digest(handle, 'sha256').hexdigest()


def read(path):
    return json.loads(pathlib.Path(path).read_text())


def safe(path):
    path = pathlib.Path(path)
    assert path.is_absolute() and path.resolve() == path and path.is_relative_to(task) and not path.is_symlink()
    return path


def relative(path):
    return safe(path).relative_to(data).as_posix()


def save(path, value):
    with pathlib.Path(path).open('x', encoding='ascii', newline='\n') as handle:
        json.dump(value, handle, indent=2, ensure_ascii=True, allow_nan=False)
        handle.write('\n')
        handle.flush()
        os.fsync(handle.fileno())


def snapshot(path):
    path = safe(path)
    value = path.stat()
    assert stat.S_ISREG(path.lstat().st_mode)
    return {'bytes': value.st_size, 'sha256': sha(path), 'device': value.st_dev, 'inode': value.st_ino,
        'mode': value.st_mode, 'uid': value.st_uid, 'gid': value.st_gid, 'mtime_ns': value.st_mtime_ns, 'ctime_ns': value.st_ctime_ns}


def resources():
    ram = next(int(line.split()[1]) * 1024 for line in pathlib.Path('/proc/meminfo').read_text().splitlines() if line.startswith('MemAvailable:'))
    disk = shutil.disk_usage(task)
    return {'recorded_at_utc': now(), 'available_memory_bytes': ram, 'disk_free_bytes': disk.free,
        'ram_minimum_bytes': 3 * 1024 ** 3, 'disk_minimum_bytes': 8 * 1024 ** 3,
        'sufficient': ram >= 3 * 1024 ** 3 and disk.free >= 8 * 1024 ** 3}


progress = (job / 'progress.jsonl').open('x', encoding='ascii')


def event(phase, **fields):
    progress.write(json.dumps({'phase': phase, 'recorded_at_utc': now(), **fields}) + '\n')
    progress.flush()
    os.fsync(progress.fileno())
