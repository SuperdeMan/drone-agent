"""Bind preserved real HTTP exchanges to the assembled Santolan representation.

将保留的真实 HTTP 交换绑定到组合后的 Santolan 表示。
"""
import hashlib
import json
import os
import runpy
from datetime import datetime, timezone
from pathlib import Path

OUT = Path(__file__).resolve().parent
FAST = OUT.parent
ROOT = FAST.parent
DATA = ROOT.parent.resolve()
REPO = DATA / 'p6-road-views-20261014/repo'
RECOVERY = FAST / 'santolan-recovery-v1'
SEGMENTS = RECOVERY / 'segmented-remainder-attempt-v1'
IDENT = '6024868fe316a7000568a583'
EXPECTED = '1127710341b5b785dd151e7e3fa8de9c0ab1cc3dc9f7f8b4df0acdee767be4c5'
VERIFIER = 'bd49f89a5916b6401705524990170e6224cdf485c5dc8e7016b71d42898c0c88'
CATALOG_SHA = '06e4692c5aad0b88aed870280609aafec29aa60b5b3a4b7d2c773ade914158b4'
SIZE = 253851020
now = lambda: datetime.now(timezone.utc).isoformat()


def sha(path):
    with Path(path).open('rb') as handle:
        return hashlib.file_digest(handle, 'sha256').hexdigest()


def read(path):
    return json.loads(Path(path).read_text(encoding='utf-8'))


def relative(path):
    path = Path(path).resolve()
    assert path.is_relative_to(ROOT.resolve()) and path.is_file() and not path.is_symlink()
    return path.relative_to(DATA).as_posix()


def save(name, value):
    path = OUT / name
    with path.open('x', encoding='ascii', newline='\n') as handle:
        json.dump(value, handle, indent=2, ensure_ascii=True, allow_nan=False)
        handle.write('\n')
        handle.flush()
        os.fsync(handle.fileno())
    return path


def binding(path):
    path = Path(path)
    return {'file': relative(path), 'bytes': path.stat().st_size, 'sha256': sha(path)}

    body_paths.append(body)
