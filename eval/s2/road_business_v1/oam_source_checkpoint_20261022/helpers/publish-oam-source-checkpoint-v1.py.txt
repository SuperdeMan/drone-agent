def sensitive(path):
    if 'ssh' in path.name.lower() and 'stderr' in path.name.lower():return 'raw_ssh_stderr'
    text=path.read_bytes().decode('utf-8',errors='replace')
    if re.search(r'https?://(?:login\.|[^/\s]*tailscale[^/\s]*/(?:a|auth|login))',text,re.I):return 'connection_login_url'
    if path.suffix=='.json':
        def unsafe(value):
            if isinstance(value,dict):
                for key,item in value.items():
                    norm=key.lower().replace('-','_')
                    if norm in {'stderr','raw_stderr','stderr_b64','stderr_text','ssh_stderr'} and isinstance(item,str):return True
                    if norm in {'authorization','proxy_authorization','token','access_token','refresh_token','id_token','api_key','client_secret','secret_key'} and isinstance(item,str) and item.strip():return True
                    if unsafe(item):return True
            elif isinstance(value,list):return any(unsafe(item) for item in value)
            return False
        if unsafe(json.loads(text)):return 'raw_connection_output_or_authorization_field'
    return None
