def read(path):
    return json.loads(Path(path).read_text(encoding='utf-8'))


def sha(path):
    with Path(path).open('rb') as handle:
        return hashlib.file_digest(handle, 'sha256').hexdigest()


def save(path, value):
    with Path(path).open('x', encoding='ascii', newline='\n') as handle:
        json.dump(value, handle, indent=2, ensure_ascii=True, allow_nan=False)
        handle.write('\n')
        handle.flush()
        os.fsync(handle.fileno())


def remote_path(path):
    return str(PurePosixPath(TASK).parent / Path(path).resolve().relative_to(DATA).as_posix())


def bindings(paths):
    return {p.relative_to(DATA).as_posix(): {'bytes': p.stat().st_size, 'sha256': sha(p)} for p in sorted(paths)}


def archive(path, paths):
    with path.open('xb') as out, tarfile.open(fileobj=out, mode='w') as bundle:
        for p in sorted(paths):
            bundle.add(p, arcname=p.relative_to(DATA).as_posix(), recursive=False)

def transport(command, label, **kwargs):
    started = now()
    child = subprocess.run(command, capture_output=True, **kwargs)
    receipt = {'started_at_utc': started, 'finished_at_utc': now(), 'returncode': child.returncode,
        'stdout_bytes': len(child.stdout), 'stdout_sha256': hashlib.sha256(child.stdout).hexdigest(),
        'stderr_bytes_withheld': len(child.stderr), 'stderr_sha256': hashlib.sha256(child.stderr).hexdigest(),
        'ssh_stderr_not_printed': True}
    save(CONTROL / (label + '-receipt.json'), receipt)
    if label == 'execution':
        with (CONTROL / 'execution-stdout.json').open('xb') as handle:
            handle.write(child.stdout)
    if child.returncode:
        raise RuntimeError('Transport failed at ' + label + '; actual receipt retained, no overwrite retry')
    return child
