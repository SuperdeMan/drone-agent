def fingerprint(path, expected=None, expected_bytes=None):
    digest=sha(path);size=path.stat().st_size
    if expected is not None:assert digest==expected,str(path)
    if expected_bytes is not None:assert size==expected_bytes,str(path)
    return {'file':str(path),'sha256':digest,'bytes':size}

def sensitive_reason(path):
    if 'ssh' in path.name.lower() and 'stderr' in path.name.lower() and path.suffix=='.log':
        return 'raw_ssh_stderr_not_copied'
    text=path.read_bytes().decode('utf-8',errors='replace')
    if re.search(r'https?://(?:login\.|[^/\s]*tailscale[^/\s]*/(?:a|auth|login))',text,re.I):
        return 'potential_connection_login_url_not_copied'
    if path.suffix=='.json':
        value=json.loads(text)
        def secret(node):
            if isinstance(node,dict):
                for key,item in node.items():
                    normalized=key.lower().replace('-','_')
                    if normalized in {'authorization','proxy_authorization','token','access_token','refresh_token',
                        'id_token','api_key','client_secret','secret_key'} and isinstance(item,str) and item.strip():
                        return True
                    if secret(item):return True
            elif isinstance(node,list):return any(secret(item) for item in node)
            return False
        if secret(value):return 'potential_authorization_or_token_value_not_copied'
    return None
