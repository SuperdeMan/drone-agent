def sha(data: bytes) -> str:
    return hashlib.sha256(data).hexdigest()


def label(path: Path) -> str:
    for prefix, base in (("ROOT/", ROOT), ("REPO/", REPO)):
        if path.is_relative_to(base):
            return prefix + path.relative_to(base).as_posix()
    raise ValueError(f"Outside authorized project roots: {path}")


def walk_leaves(value, path="$", rows=None):
    if rows is None:
        rows = []
    if isinstance(value, dict):
        for key, child in value.items():
            child_path = f"{path}.{key}"
            if LEAF_PATTERN.search(key) and "overview" not in key.lower() and not isinstance(child, dict):
                if not isinstance(child, list) or len(child) <= 20:
                    rows.append({"json_path": child_path, "value": child})
            if isinstance(child, (dict, list)):
                walk_leaves(child, child_path, rows)
    elif isinstance(value, list):
        for number, child in enumerate(value):
            if isinstance(child, (dict, list)):
                walk_leaves(child, f"{path}[{number}]", rows)
    return rows
