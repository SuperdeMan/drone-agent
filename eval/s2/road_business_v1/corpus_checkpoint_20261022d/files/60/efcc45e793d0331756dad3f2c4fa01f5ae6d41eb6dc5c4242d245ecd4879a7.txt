def sha(path):
    """Stream immutable file hashes. / 流式核对不可变文件摘要。"""
    with Path(path).open("rb") as f:
        return hashlib.file_digest(f, "sha256").hexdigest()


def save(path, value):
    """Create new evidence exclusively. / 仅独占新建证据。"""
    with Path(path).open("x", encoding="ascii") as f:
        json.dump(value, f, indent=2, ensure_ascii=True, allow_nan=False)
        f.write("\n")
        f.flush()
        os.fsync(f.fileno())


def read(path):
    return json.loads(Path(path).read_text())


def safe(path):
    """Constrain new and reused paths to the existing task root. / 将新旧路径限制在既有数据根。"""
    p = Path(path)
    assert p.is_absolute() and p.resolve() == p and p.is_relative_to(task) and not p.is_symlink()
    return p


def relative(path):
    return safe(path).relative_to(data).as_posix()


def snapshot(path):
    p = safe(path)
    s = p.stat()
    assert stat.S_ISREG(p.lstat().st_mode)
    return {"bytes": s.st_size, "sha256": sha(p), "device": s.st_dev, "inode": s.st_ino,
            "mode": s.st_mode, "mtime_ns": s.st_mtime_ns, "ctime_ns": s.st_ctime_ns}


def resources():
    ram = next(int(s.split()[1])*1024 for s in Path("/proc/meminfo").read_text().splitlines() if s.startswith("MemAvailable:"))
    disk = shutil.disk_usage(task)
    return {"recorded_at_utc": now(), "memory_available_bytes": ram, "disk_free_bytes": disk.free,
            "minimum_ram_bytes": 3*1024**3, "minimum_disk_bytes": 8*1024**3,
            "sufficient": ram >= 3*1024**3 and disk.free >= 8*1024**3}
