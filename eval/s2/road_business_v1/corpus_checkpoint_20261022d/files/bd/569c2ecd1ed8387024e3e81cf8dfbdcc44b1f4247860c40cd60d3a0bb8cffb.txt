def require(condition, message):
    """Reject unsupported metadata assertions. / 拒绝无支持的元数据断言。"""
    if not condition:
        raise ValueError(message)


def read(path):
    return json.loads(Path(path).read_text(encoding="utf-8"))


def sha(path):
    with Path(path).open("rb") as stream:
        return hashlib.file_digest(stream, "sha256").hexdigest()


def canonical(value):
    encoded = json.dumps(value, sort_keys=True, ensure_ascii=False, separators=(",", ":"), allow_nan=False)
    return hashlib.sha256(encoded.encode()).hexdigest()


def resolve(value):
    """Map declared cloud data paths to local metadata. / 将云端数据路径映射到本地元数据。"""
    value = str(value).replace("\\", "/")
    for prefix in ("D:/drone-agent-data/", "/home/ubuntu/drone-agent-data/p6-corpus-completion-20261018-cloud/data/"):
        if value.startswith(prefix):
            value = value[len(prefix):]
            break
    path = Path(value) if Path(value).is_absolute() else DATA / value
    path = path.resolve()
    require(path.is_relative_to(ROOT) or path.is_relative_to(REPO), "input path outside authorized roots")
    return path


def bind(value, expected=None, size=None):
    path = resolve(value)
    require(path.is_file(), f"input file missing: {path}")
    digest = sha(path)
    require(expected is None or digest == expected, f"input SHA mismatch: {path}")
    require(size is None or path.stat().st_size == size, f"input size mismatch: {path}")
    prior = BINDINGS.get(str(path))
    require(prior is None or prior["sha256"] == digest, f"input changed during audit: {path}")
    BINDINGS[str(path)] = {"file": str(path), "sha256": digest, "bytes": path.stat().st_size}
    return path


def unique(rows, key="sample_id"):
    mapped = {row[key]: row for row in rows}
    require(len(mapped) == len(rows), f"duplicate {key}")
    return mapped


def footprint(receipt):
    """Recompute complete native-image envelopes from actual affine/CRS.

从实际仿射与坐标系重算完整原生图像包络，绝不使用目标裁边包络。
"""
    width, height = receipt["width"], receipt["height"]
    require(type(width) is int and type(height) is int and width > 0 and height > 0, "invalid native dimensions")
    col, row, win_width, win_height = receipt["native_window_xywh"]
    require(all(type(v) is int for v in (col, row, win_width, win_height)), "noninteger native window")
    require((width, height) == (win_width, win_height), "native window dimensions differ")
    raster = receipt["raster"]
    require(0 <= col < col + width <= raster["width"] and 0 <= row < row + height <= raster["height"], "window outside source raster")
    source_affine = Affine(*raster["transform"][:6])
    window_affine = Affine(*receipt["window_transform"][:6])
    calculated = source_affine * Affine.translation(col, row)
    require(all(abs(a - b) <= 1e-7 for a, b in zip(calculated, window_affine)), "source/window affine mismatch")
    require(abs(window_affine.determinant) > 0, "singular native affine")
    corners = [window_affine * point for point in ((0, 0), (width, 0), (width, height), (0, height))]
    bounds = [min(x[0] for x in corners), min(x[1] for x in corners), max(x[0] for x in corners), max(x[1] for x in corners)]
    geographic21 = list(transform_bounds(raster["crs"], "EPSG:4326", *bounds, densify_pts=21))
    left, bottom, right, top = bounds
    points = []
    for index in range(1001):
        fraction = index / 1000
        points.extend(((left + (right - left) * fraction, bottom), (left + (right - left) * fraction, top),
                       (left, bottom + (top - bottom) * fraction), (right, bottom + (top - bottom) * fraction)))
    xs, ys = transform(raster["crs"], "EPSG:4326", [p[0] for p in points], [p[1] for p in points])
    geographic1001 = [min(xs), min(ys), max(xs), max(ys)]
    require(all(math.isfinite(v) for v in bounds + geographic21 + geographic1001), "nonfinite geographic envelope")
    require(geographic1001[2] - geographic1001[0] < 1, "unexpected antimeridian or oversized native view")
    combined = [min(geographic21[0], geographic1001[0]), min(geographic21[1], geographic1001[1]),
                max(geographic21[2], geographic1001[2]), max(geographic21[3], geographic1001[3])]
    envelope = [math.nextafter(v, -math.inf if index < 2 else math.inf) for index, v in enumerate(combined)]
    return {"sample_id": receipt["sample_id"], "source_crs": raster["crs"], "native_window_xywh": receipt["native_window_xywh"],
            "source_affine": raster["transform"], "native_window_affine": receipt["window_transform"], "native_image_corners": corners,
            "native_image_bounds": bounds, "transform_bounds_densify21_lon_lat": geographic21,
            "independent_1001_per_edge_bounds_lon_lat": geographic1001, "conservative_full_image_envelope_lon_lat": envelope,
            "complete_native_derived_image_not_target_polygon_or_parent_raster": True}


def check_derivation(sample, remote_rows):
    """Bind local metadata or exact already-executed remote byte evidence.

绑定本地元数据，或绑定已实际执行的逐文件远端字节证据；本审核不联网。
"""
    details = []
    for key, value in sample["derivation"].items():
        if not key.endswith("_file"):
            continue
        digest = sample["derivation"][key[:-5] + "_sha256"]
        path = resolve(value)
        if path.is_file():
            bind(path, digest)
            details.append({"declared_file": value, "sha256": digest, "mode": "local_bytes_rehashed"})
        else:
            mirror = ROOT / "cloud-final-source-evidence-v1" / path.relative_to(ROOT)
            if mirror.is_file():
                bind(mirror, digest)
                details.append({"declared_file": value, "sha256": digest, "local_mirror_file": str(mirror),
                                "mode": "exact_digest_preserved_local_metadata_mirror; current_cdf_execution_bound_original_cloud_path"})
                continue
            witness = remote_rows.get(str(path))
            require(witness is not None and witness["passed"] is True and witness["sha256"] == digest
                    and sample["sample_id"] in witness["sample_ids"], f"missing local/remote derivation binding: {value}")
            REMOTE_BINDINGS[str(path)] = {"declared_file": witness["file"], "sha256": digest, "bytes": witness["bytes"],
                                         "basis": "actual8354-byte-result; original files remain cloud-resident; no current local reread claimed"}
            details.append({"declared_file": value, "sha256": digest, "mode": "prior_actual_cloud_byte_result_plus_current_native_source_execution"})
    capture = sample["capture_record"]
    inventory = read(bind(capture["inventory_file"], capture["inventory_sha256"]))
    record = unique(inventory["rows"], "source_observation_id")[sample["sample_id"]]
    require(record["image_sha256"] == sample["image_sha256"] and record["derivation_sha256"] == canonical(sample["derivation"]), "capture inventory binding mismatch")
    require(record["observed"] == sample["observed"], "capture observed grouping mismatch")
    return details

def load_part(directory, numbers, boundary):
    """Bind a completed annotation-provenance part. / 绑定完整的标注溯源分部。"""
    result = read(bind(directory / "result.json"))
    corpus = read(bind(directory / "reviewed-candidates.json"))
    protocol = read(bind(directory / "protocol-annotation-only.json"))
    expected = sum(EXPECTED[number] for number in numbers)
    require(result["status"] == "passed" and result["checked"] == result["passed"] == len(result["rows"]) == expected, "annotation provenance part incomplete")
    require(not result["scope_errors"] and not result["pending_protocol_reverification"] and not result["historical_unmatched"], "annotation provenance part has unresolved issues")
    require(all(not row["issues"] for row in result["rows"]), "annotation provenance row issues")
    require(set(result["only_reviewers"]) == {f"bounded_verifier_{number}" for number in numbers}, "annotation part reviewer set differs")
    require(result["source_sha256"] == CORE["p6_business_eval.py"], "annotation source core differs")
    require(protocol["target_boundary_policy"] == boundary, "approved boundary policy changed")
    require(corpus["formal_release_eligible"] is False and corpus["grouping_verified"] is False, "annotation part already asserts grouping/release")
    candidates = {row["sample"]["sample_id"]: row for row in corpus["rows"]}
    require(len(candidates) == expected and set(candidates) == set(unique(result["rows"])), "annotation candidate/result ID mismatch")
    mirrors = {}
    for number in numbers:
        reviewer = f"bounded_verifier_{number}"
        mirror = bind(directory / f"{reviewer}.json")
        bind(ROOT / "verification" / reviewer / "reviews.json", sha(mirror))
        mirrors[reviewer] = mirror
    return {"directory": str(directory), "candidates": candidates, "protocol": protocol, "mirrors": mirrors}

            require(native_sample["image_sha256"] == receipt["image_sha256"] and native_sample["derivation"]["extraction_sha256"] == sha(receipt_path)
                    and resolve(native_sample["derivation"]["extraction_file"]) == receipt_path, "current cdf source execution not bound to this receipt")
            require(native_sample["width"] == receipt["width"] and native_sample["height"] == receipt["height"]
                    and native_sample["capture_record"]["parent_sha256"] == receipt.get("parent_sha256", receipt.get("source_published_lfs_sha256")), "source native dimensions/parent identity differs")
            case["source_derivation_bindings"] = check_derivation(native_sample, sources["remote"])
            group = receipt["asset_group_id"]
            observed = native_sample["observed"]
            require(observed["asset_id"] == observed["near_duplicate_group"] == group, "existing group/near-duplicate rule differs")
            if kind == "crasar":
                site = rules["crasar_site_map"][receipt["source_orthomosaic"]]
                require(observed["site_id"] == site["site_id"] and observed["capture_batch_id"] == "author-parent-" + receipt["source_published_lfs_sha256"], "CRASAR site/capture grouping differs")
                require(receipt["proposed_partition"] == site["planned_partition"], "CRASAR registered partition differs")
            else:
                require(observed["site_id"] == "oam-site-" + receipt["catalog_object_id"]
                        and observed["capture_batch_id"] == "publisher-object-" + receipt["catalog_object_id"], "OAM site/capture grouping differs")
                require(receipt["proposed_partition"] == "calibration_candidate" and receipt["repeated_capture_eligible"] is False, "OAM preregistered partition/repeat eligibility differs")
                site = {"site_id": observed["site_id"], "catalog_object_id": receipt["catalog_object_id"], "planned_partition": "calibration_candidate"}
            case.update(proposed_group_id=group, proposed_observed_grouping=observed, registered_site=site,
                        preregistered_partition=receipt["proposed_partition"], source_kind=kind, source_candidate_commit=CANDIDATE,
                        final_row=final, final_row_sha256=actual["final_row_sha256"], extraction_file=str(receipt_path),
                        extraction_sha256=sha(receipt_path), fixed_world_scope=scope, fixed_scope_canonical_sha256=canonical(scope),
                        footprint=footprint(receipt), source_only_exposure_placeholder_preserved=(kind == "oam"), formal_exposure_adjudicated=False)
            for key in FIELDS:
                if finding[key] is not True:
                    case["hold_reasons"].append(f"{key} is not explicitly true")
            polygon = final["target_polygon"]
            if polygon is None:
                case["hold_reasons"].append("complete target_polygon remains absent")
            else:
                require(isinstance(polygon, list) and len(polygon) >= 3 and all(isinstance(point, list) and len(point) == 2
                        and all(type(value) in (int, float) and math.isfinite(value) and 0 <= value <= 1 for value in point) for point in polygon), "target polygon is not finite normalized coordinates")
                geometry = Polygon(polygon)
                require(geometry.is_valid and geometry.area > 0, "target polygon is not a valid nonzero polygon")
                case["target_polygon_geometrically_valid"] = True
            if final["verdict"] == "excluded_source":
                case["hold_reasons"].append("excluded_source is not an admissible physical road target")
            case["mechanical_binding_supported"] = True
        except Exception as exc:
            case["mechanical_binding_supported"] = False
            case["binding_issues"].append(f"{type(exc).__name__}: {exc}")
            case["hold_reasons"].append("binding evidence not established")
        cases.append(case)
    return cases
