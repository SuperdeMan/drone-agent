def locate(name):

    if direct.is_file():

    return None


def add(value, expected=None, size=None):
    name = relative(value)
    local = locate(name)
    if local is not None:
        assert not local.is_symlink()
        actual = sha(local)
        assert expected is None or actual == expected, 'Local dependency hash differs: ' + name
        assert size is None or local.stat().st_size == size
        expected, size = actual, local.stat().st_size
        LOCAL[name] = local
        if local.resolve() != (DATA / name).resolve():
            ALIASES.append({'declared_file': name, 'exact_local_mirror_file': str(local),
                            'sha256': expected, 'bytes': size, 'declared_path_not_rewritten': True})
    assert C['sha256_value'](expected), 'Missing exact dependency SHA: ' + name
    old = BINDINGS.get(name)
    assert old is None or old['sha256'] == expected, 'Conflicting exact dependency: ' + name
    if old is not None and size is None:
        size = old['bytes']
    BINDINGS[name] = {'sha256': expected, 'bytes': size}
    return local


def metadata(value, expected=None):
    path = add(value, expected)
    assert path is not None, 'Necessary metadata needs an exact read-only mirror: ' + relative(value)
    return read(path)

def ssh(code, phase, timeout=1200):
    started = now()
    try:
        child = subprocess.run(['ssh', *connection.arguments(), connection.target, REMOTE_PYTHON + ' -B -u -'],
            input=code.encode('utf-8'), capture_output=True, timeout=timeout)
    except subprocess.TimeoutExpired as exc:
        save(CONTROL / (phase + '-transport-timeout.json'), {'started_at': started, 'finished_at': now(),
            'timeout_seconds': timeout, 'stdout_bytes': len(exc.stdout or b''),
            'stdout_sha256': hashlib.sha256(exc.stdout or b'').hexdigest(),
            'stderr_bytes': len(exc.stderr or b''), 'stderr_sha256': hashlib.sha256(exc.stderr or b'').hexdigest(),
            'raw_ssh_stderr_printed': False, 'automatic_rerun_attempted': False})
        raise RuntimeError('Transport timeout; only byte counts and hashes retained; inspect exact remote state before any retry.') from None
    receipt = {'started_at': started, 'finished_at': now(), 'returncode': child.returncode,
        'stdout_bytes': len(child.stdout), 'stdout_sha256': hashlib.sha256(child.stdout).hexdigest(),
        'stderr_bytes': len(child.stderr), 'stderr_sha256': hashlib.sha256(child.stderr).hexdigest(),
        'raw_ssh_stderr_printed': False}
    save(CONTROL / (phase + '-transport-receipt.json'), receipt)
    if child.returncode != 0:
        raise RuntimeError('Remote ' + phase + ' failed; SSH diagnostics withheld, receipt retained.')
    try:
        return json.loads(child.stdout)
    except (ValueError, UnicodeError):
        raise RuntimeError('Remote ' + phase + ' did not return strict JSON; diagnostics withheld.') from None
