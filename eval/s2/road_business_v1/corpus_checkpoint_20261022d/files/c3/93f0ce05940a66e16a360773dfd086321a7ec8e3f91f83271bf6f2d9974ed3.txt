now = lambda: datetime.now(timezone.utc).isoformat()


def sha(path):
    with Path(path).open("rb") as f:
        return hashlib.file_digest(f, "sha256").hexdigest()


def read(path):
    return json.loads(Path(path).read_text(encoding="utf-8"))


def save(path, value):
    with Path(path).open("x", encoding="ascii") as f:
        json.dump(value, f, indent=2, ensure_ascii=True, allow_nan=False)
        f.write("\n")
        f.flush()
        os.fsync(f.fileno())


def remote(path):
    return TASK+"/"+Path(path).resolve().relative_to(ROOT.resolve()).as_posix()


def transport(arguments, label, **kwargs):
    """Withhold raw SSH stderr and auth diagnostics. / 不输出原SSH stderr或认证诊断。"""
    started = now()
    child = subprocess.run(arguments, capture_output=True, **kwargs)
    receipt = {"started_at_utc": started, "finished_at_utc": now(), "returncode": child.returncode,
        "stdout_bytes": len(child.stdout), "stdout_sha256": hashlib.sha256(child.stdout).hexdigest(),
        "stderr_bytes_withheld": len(child.stderr), "stderr_sha256": hashlib.sha256(child.stderr).hexdigest(),
        "raw_ssh_stderr_or_login_url_printed": False}
    save(CONTROL/(label+"-receipt.json"), receipt)
    if child.returncode:
        raise RuntimeError("transport_nonzero_"+label)
    return child
