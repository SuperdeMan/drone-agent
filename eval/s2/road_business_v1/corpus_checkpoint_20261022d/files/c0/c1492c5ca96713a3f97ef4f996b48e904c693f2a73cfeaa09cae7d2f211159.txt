def sha(path):
    with Path(path).open('rb') as stream:
        return hashlib.file_digest(stream, 'sha256').hexdigest()


def read(path):
    return json.loads(Path(path).read_text(encoding='utf-8'))


def save(path, value):
    with Path(path).open('x', encoding='utf-8', newline='\n') as stream:
        json.dump(value, stream, ensure_ascii=False, indent=2, allow_nan=False)
        stream.write('\n')


def rel(path):
    return Path(path).resolve().relative_to(DATA.resolve()).as_posix()


def only_context_diff(before, after):
    first, second = copy.deepcopy(before), copy.deepcopy(after)
    for row in (first, second):
        for key in ('context_file', 'context_sha256', 'verification_context_file', 'verification_context_sha256'):
            row['ai_assistance'].pop(key, None)
    return first == second
