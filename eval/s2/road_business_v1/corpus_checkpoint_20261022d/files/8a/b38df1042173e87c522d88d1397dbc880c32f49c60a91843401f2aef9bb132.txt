def digest(data):
    return hashlib.sha256(data).hexdigest()


def file_digest(path):
    h = hashlib.sha256()
    with path.open("rb") as stream:
        for block in iter(lambda: stream.read(1024 * 1024), b""):
            h.update(block)
    return h.hexdigest()

def safe_rows(value):
    # Drop every unlisted key before any further processing or output. / 先丢弃白名单外的全部字段。
    rows = value.get("rows", []) if isinstance(value, dict) else value
    return [{k: v for k, v in row.items() if k in SAFE} for row in rows]


def local_xy(gps):
    lat, lon = gps
    return ((lon - ORIGIN[1]) * 111320.0 * math.cos(math.radians(ORIGIN[0])),
            (lat - ORIGIN[0]) * 110946.0)


def good_gps(value):
    return (isinstance(value, list) and len(value) == 2
            and all(isinstance(v, (float, int)) and math.isfinite(v) for v in value)
            and -90 <= value[0] <= 90 and -180 <= value[1] <= 180)


def parse_clock(value):
    try:
        return datetime.strptime(value, "%Y:%m:%d %H:%M:%S")
    except (TypeError, ValueError):
        return None


def parse_tiff(data):
    if len(data) < 8 or data[:2] not in (b"II", b"MM"):
        raise ValueError("invalid EXIF TIFF header")
    endian = "<" if data[:2] == b"II" else ">"

    def unpack(fmt, offset):
        return struct.unpack_from(endian + fmt, data, offset)

    if unpack("H", 2)[0] != 42:
        raise ValueError("unsupported EXIF TIFF")
    sizes = {1: 1, 2: 1, 3: 2, 4: 4, 5: 8, 7: 1, 9: 4, 10: 8}

    def ifd(offset):
        if not isinstance(offset, int) or offset <= 0 or offset + 2 > len(data):
            return {}
        count = unpack("H", offset)[0]
        if count > 1000 or offset + 2 + 12 * count > len(data):
            raise ValueError("invalid EXIF IFD extent")
        values = {}
        for number in range(count):
            entry = offset + 2 + number * 12
            tag, kind, n = unpack("HHI", entry)
            if kind not in sizes or n > 100000:
                continue
            length = sizes[kind] * n
            pos = entry + 8 if length <= 4 else unpack("I", entry + 8)[0]
            if pos < 0 or pos + length > len(data):
                continue
            raw = data[pos:pos + length]
            if kind == 2:
                value = raw.decode("ascii", errors="replace").strip("\x00 ")
            elif kind in (5, 10):
                fmt = "II" if kind == 5 else "ii"
                values_n = [unpack(fmt, pos + j * 8) for j in range(n)]
                value = [a / b if b else None for a, b in values_n]
                value = value[0] if n == 1 else value
            elif kind in (1, 3, 4, 9):
                fmt = {1: "B", 3: "H", 4: "I", 9: "i"}[kind]
                value = list(unpack(fmt * n, pos))
                value = value[0] if n == 1 else value
            else:
                continue
            values[tag] = value
        return values

    top = ifd(unpack("I", 4)[0])
    sub = ifd(top.get(34665))
    gps_ifd = ifd(top.get(34853))
    gps = None
    if all(k in gps_ifd for k in (1, 2, 3, 4)):
        def degrees(v):
            return float(v[0]) + float(v[1]) / 60 + float(v[2]) / 3600
        gps = [degrees(gps_ifd[2]) * (-1 if gps_ifd[1] == "S" else 1),

def jpeg_header(raw):
    # Marker and TIFF parsing only; no image library, RGB, entropy decode or render. / 仅解析标记与 TIFF，不解码或渲染像素。
    if raw[:2] != b"\xff\xd8":
        raise ValueError("not a JPEG original")
    out = {"primary_sos_seen": False, "terminal_eoi_present": raw[-2:] == b"\xff\xd9"}
    pos = 2
    xmp = {}
    while pos < len(raw):
        if raw[pos] != 255:
            raise ValueError("invalid JPEG marker boundary")
        while pos < len(raw) and raw[pos] == 255:
            pos += 1
        marker = raw[pos]
        pos += 1
        if marker == 218:
            out["primary_sos_seen"] = True
            break
        if marker in (216, 217, 1) or 208 <= marker <= 215:
            continue
        length = struct.unpack_from(">H", raw, pos)[0]
        payload = raw[pos + 2:pos + length]
        if length < 2 or pos + length > len(raw):
            raise ValueError("truncated JPEG metadata segment")
        if marker == 225 and payload.startswith(b"Exif\x00\x00"):
            out.update(parse_tiff(payload[6:]))
        if marker == 225 and b"<x:xmpmeta" in payload:
            text = payload.decode("utf-8", errors="replace")
            for key in ("RelativeAltitude", "AbsoluteAltitude", "GimbalYawDegree",
                        "GimbalPitchDegree", "GimbalRollDegree", "FlightYawDegree"):
                match = re.search(r"(?:[:\s])" + key + r'="([^"]+)"', text)
                if match:
                    try:
                        value = float(match.group(1))
                        if math.isfinite(value):
                            xmp[key] = value
                    except ValueError:
                        pass
        if marker in {192, 193, 194, 195, 197, 198, 199, 201, 202, 203, 205, 206, 207}:
            out["height_px"], out["width_px"] = struct.unpack_from(">HH", payload, 1)
        pos += length
    out["metadata_header_bytes"] = pos
    out["xmp_pose"] = xmp
    return out


def footprint(meta):
    pose = meta.get("xmp_pose", {})
    required = ("RelativeAltitude", "GimbalYawDegree", "GimbalPitchDegree", "GimbalRollDegree")
    if not good_gps(meta.get("gps")):
        return None, "gps_unknown"
    if not all(k in pose for k in required) or not meta.get("focal35"):
        return None, "camera_pose_or_focal35_unknown"
    if meta.get("orientation", 1) != 1:
        return None, "unsupported_orientation"
    width, height = meta.get("width_px"), meta.get("height_px")
    if not width or not height or pose["RelativeAltitude"] <= 0:
        return None, "invalid_dimensions_or_height"
    focal = float(meta["focal35"]) / 43.2666 * math.hypot(width, height)
    yaw, pitch, roll = [math.radians(pose[k]) for k in required[1:]]
    forward = (math.sin(yaw) * math.cos(pitch), math.cos(yaw) * math.cos(pitch), math.sin(pitch))
    right0 = (math.cos(yaw), -math.sin(yaw), 0.0)
    down0 = (-forward[2] * right0[1], forward[2] * right0[0],
             forward[0] * right0[1] - forward[1] * right0[0])
    right = tuple(math.cos(roll) * a + math.sin(roll) * b for a, b in zip(right0, down0))
    down = tuple(-math.sin(roll) * a + math.cos(roll) * b for a, b in zip(right0, down0))
    east, north = local_xy(meta["gps"])
    points = []
    for u, v in ((0, 0), (1, 0), (1, 1), (0, 1)):
        x, y = (u - 0.5) * width / focal, (v - 0.5) * height / focal
        dx, dy, dz = (f + x * r + y * d for f, r, d in zip(forward, right, down))
        if dz >= -1e-6:
            return None, "full_frame_corner_does_not_intersect_ground"
        t = pose["RelativeAltitude"] / -dz
        if t * math.hypot(dx, dy) > 400:
            return None, "full_frame_corner_exceeds_existing_400m_range"
        points.append((east + t * dx, north + t * dy))
    poly = Polygon(points)
    if not poly.is_valid or poly.area <= 0:
        return None, "invalid_approximate_footprint"
    return poly, "approximate_full_frame_ground_intersection"


class Union:
    def __init__(self, ids):
        self.parent = {i: i for i in ids}

    def find(self, value):
        while self.parent[value] != value:
            self.parent[value] = self.parent[self.parent[value]]
            value = self.parent[value]
        return value

    def link(self, a, b):
        a, b = self.find(a), self.find(b)
        if a != b:
            self.parent[max(a, b)] = min(a, b)

    def groups(self):
        out = defaultdict(list)
        for ident in self.parent:
            out[self.find(ident)].append(ident)
        return {k: sorted(v) for k, v in sorted(out.items())}
