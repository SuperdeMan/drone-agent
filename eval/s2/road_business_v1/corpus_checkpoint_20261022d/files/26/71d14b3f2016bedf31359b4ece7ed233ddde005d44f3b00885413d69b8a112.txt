def digest(path):
    """Hash encoded bytes without decoding images. / 只哈希编码字节，不解码图片。"""
    path = Path(path)
    key = (str(path), path.stat().st_size, path.stat().st_mtime_ns)
    if key not in HASH_CACHE:
        h = sha256()
        with path.open("rb") as f:
            while block := f.read(4 * 1024 * 1024):
                h.update(block)
        HASH_CACHE[key] = h.hexdigest()
    return HASH_CACHE[key]


def binding(path, expected=None):
    """Bind an actual file and retain a missing file explicitly. / 绑定实存文件，明确保留缺失。"""
    path = Path(path)
    result = {"file": str(path), "exists": path.is_file()}
    if result["exists"]:
        result.update(bytes=path.stat().st_size, sha256=digest(path))
    if expected:
        result.update(expected_sha256=expected, sha256_matches=result.get("sha256") == expected)
    READ_BINDINGS[str(path)] = result
    return result.copy()


def read(path):
    """Read a JSON container; callers select metadata only. / 读取JSON容器，由调用方只选元数据。"""
    path = Path(path)
    binding(path)
    return json.loads(path.read_text(encoding="utf-8-sig"))


def local_path(declared):
    """Map the documented data-root prefix without changing source files. / 只映射已知数据根前缀，不改来源文件。"""
    normalized = str(declared).replace("\\", "/")
    marker = ROOT.name + "/"
    if marker in normalized:
        return ROOT / normalized.split(marker, 1)[1]
    return Path(declared)


def resolve(declared, expected):
    """Find the original or its known byte-identical metadata mirror. / 查原路径及已知同字节元数据镜像。"""
    primary = local_path(declared)
    possibilities = [primary]
    if primary.is_relative_to(ROOT):
        possibilities.append(ROOT / "cloud-final-source-evidence-v1" / primary.relative_to(ROOT))
    found = [p for p in possibilities if p.is_file() and digest(p) == expected]
    record = {"declared_file": str(declared), "declared_local_file": str(primary),
              "expected_sha256": expected, "declared_local_exists": primary.is_file()}
    if not found:
        record.update(exists=False, sha256_matches=False)
        return None, record
    actual = found[0]
    record.update(binding(actual, expected), path_relocated=actual != primary)
    return actual, record


def require_remote(declared, expected, size, role, observation=None):
    """Record exact existing-cloud checks; do not contact the cloud. / 记录云端待核精确清单，不连接云端。"""
    normalized = str(declared).replace("\\", "/")
    marker = ROOT.name + "/"
    remote = REMOTE + "/" + normalized.split(marker, 1)[1]
    if remote in REMOTE_REQUIREMENTS:
        record = REMOTE_REQUIREMENTS[remote]
        assert record["expected_sha256"] == expected
        if size is not None:
            assert record["expected_bytes"] in (None, size)
            record["expected_bytes"] = size
    else:
        record = {"file": remote, "expected_sha256": expected, "expected_bytes": size,
                  "roles": [], "sample_ids": []}
        REMOTE_REQUIREMENTS[remote] = record
    if role not in record["roles"]:
        record["roles"].append(role)
    if observation and observation not in record["sample_ids"]:
        record["sample_ids"].append(observation)
    return record


def author_date(source):
    """Use agreeing author dates without inventing flight identity. / 采用一致作者日期，不虚构飞行身份。"""
    try:
        table = datetime.strptime(source["author_statistics"]["Date (mm/dd/yyy)"], "%m/%d/%Y").date()
        code = datetime.strptime(source["author_code_date_mm_dd_yyyy"], "%m-%d-%Y").date()
        if table == code and source.get("author_code_date_marked_double_check") is False:
            return table.isoformat()
    except (KeyError, TypeError, ValueError):
        pass
    return None


def same_assignment(first, current):
    """Compare image, fixed target and actual display metadata. / 比较图片、固定目标及实际显示元数据。"""
    if (first.get("image_sha256") != current["image_sha256"]
            or first.get("fixed_world_scope") != current["world_scope"]):
        return False
    keys = ("target_endpoints_normalized", "centerline_segments_normalized")
    if all(first.get(k) == current[k] for k in keys):
        return True
    bridge = ROOT / "frames-author-aligned-v4/receipts" / (current["sample_id"] + ".json")
    if not bridge.is_file():
        return False
    other = read(bridge)
    if (other["image_sha256"] != current["image_sha256"] or other["world_scope"] != current["world_scope"]
            or any(other[k] != current[k] for k in keys)):
        return False
    old_guide = local_path(first["guide_file"])
    bridge_guide = ROOT / "frames-author-aligned-v4/guides" / (current["sample_id"] + ".jpg")
    return (old_guide.is_file() and bridge_guide.is_file()
            and digest(old_guide) == other["guide_sha256"] == digest(bridge_guide))


def save(name, document):
    """Create a new report, refusing to rewrite evidence. / 新建报告，拒绝覆盖证据。"""
    with (OUT / name).open("x", encoding="utf-8") as f:
        json.dump(document, f, ensure_ascii=False, indent=2)
        f.write("\n")
