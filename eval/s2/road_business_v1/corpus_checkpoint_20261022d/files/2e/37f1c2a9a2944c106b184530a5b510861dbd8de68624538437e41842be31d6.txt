"""Seal local artifacts and add exact review metadata to the new cloud directory.

封存本地产物，并向新云端目录补入同摘要审查元数据。
"""
import base64
import hashlib
import json
import subprocess
import sys
from datetime import datetime, timezone
from pathlib import Path

sys.dont_write_bytecode = True
OUT = Path(__file__).resolve().parent
REPO = OUT.parent.parent.parent / 'p6-road-views-20261014/repo'
REMOTE = '/home/ubuntu/drone-agent-data/p6-corpus-completion-20261018-cloud/data/p6-corpus-completion-20261018/resume-20261022-next/composed22-v1'


def sha(path):
    with path.open('rb') as stream:
        return hashlib.file_digest(stream, 'sha256').hexdigest()


files = sorted(path for path in OUT.rglob('*') if path.is_file())
assert not (OUT / 'SHA256SUMS.txt').exists() and not (OUT / 'final-cloud-metadata-sync-receipt.json').exists()
with (OUT / 'SHA256SUMS.txt').open('x', encoding='ascii', newline='\n') as stream:
    for path in files:
        stream.write(sha(path) + '  ' + path.relative_to(OUT).as_posix() + '\n')
files.append(OUT / 'SHA256SUMS.txt')
metadata = [path for path in files if path.suffix.lower() in {'.json', '.py', '.md', '.txt', '.log'}]
assert all(path.is_relative_to(OUT) and not path.is_symlink() for path in metadata)
payload = {}
for path in metadata:
    blob = path.read_bytes()
    payload[path.relative_to(OUT).as_posix()] = {'sha256': hashlib.sha256(blob).hexdigest(),
        'bytes': len(blob), 'data': base64.b64encode(blob).decode()}
code = 'ROOT=' + repr(REMOTE) + '\nFILES=' + repr(payload) + '\n' + r'''
import base64,hashlib,json,pathlib
from datetime import datetime,timezone
root=pathlib.Path(ROOT).resolve()
def sha(p):
 with p.open('rb') as s:return hashlib.file_digest(s,'sha256').hexdigest()
assert sha(root/'actual-audit-v1/core-audit.json')=='667a1603811f2716b710cd9c981da9df4e146905a709c6a3b2fecbdd6d57f2c4'
rows=[]
for name,item in FILES.items():
 p=(root/name).resolve();assert p.is_relative_to(root)
 blob=base64.b64decode(item['data'],validate=True)
 assert len(blob)==item['bytes'] and hashlib.sha256(blob).hexdigest()==item['sha256']
 if p.exists():assert p.is_file() and sha(p)==item['sha256'],'Differing review artifact; never overwrite'
for name,item in FILES.items():
 p=(root/name).resolve();added=not p.exists()
 if added:
  p.parent.mkdir(parents=True,exist_ok=True)
  with p.open('xb') as s:s.write(base64.b64decode(item['data'],validate=True))
 assert sha(p)==item['sha256'] and p.stat().st_size==item['bytes']
 rows.append({'file':name,'sha256':item['sha256'],'bytes':item['bytes'],'new_exact_metadata_copy':added})
receipt={'recorded_at_utc':datetime.now(timezone.utc).isoformat(),'cloud_directory':str(root),'files':rows,
 'new_files':sum(r['new_exact_metadata_copy'] for r in rows),'all_metadata_files_identical_after':True,
 'source_tiff_or_cache_uploaded':False,'existing_files_overwritten':False,'core_audit_not_rerun':True,
 'model_calls':0,'pixels_visually_viewed':False}
with (root/'cloud-final-metadata-sync-receipt.json').open('x') as s:json.dump(receipt,s,indent=2)
print(json.dumps(receipt))
'''
compile(code, 'final-cloud-metadata-sync.py', 'exec')
sys.path.insert(0, str(REPO / 'scripts'))
from dev_stack import Connection
connection = Connection.from_environment()
started = datetime.now(timezone.utc).isoformat()
try:
    child = subprocess.run(['ssh', *connection.arguments(), connection.target, 'python3 -B -u -'],
        input=code.encode(), capture_output=True, timeout=600)
except subprocess.TimeoutExpired as exc:
    receipt = {'started_at_utc': started, 'finished_at_utc': datetime.now(timezone.utc).isoformat(),
        'timeout': True, 'stderr_bytes': len(exc.stderr or b''), 'stderr_sha256': hashlib.sha256(exc.stderr or b'').hexdigest(),
        'stdout_bytes': len(exc.stdout or b''), 'stdout_sha256': hashlib.sha256(exc.stdout or b'').hexdigest(),
        'raw_ssh_stderr_printed': False, 'automatic_retry': False}
    with (OUT / 'final-cloud-metadata-sync-timeout.json').open('x', encoding='utf-8') as stream:
        json.dump(receipt, stream, indent=2)
    raise RuntimeError('Metadata sync timeout; inspect original cloud state before retry.') from None
receipt = {'started_at_utc': started, 'finished_at_utc': datetime.now(timezone.utc).isoformat(),
    'returncode': child.returncode, 'stderr_bytes': len(child.stderr), 'stderr_sha256': hashlib.sha256(child.stderr).hexdigest(),
    'stdout_bytes': len(child.stdout), 'stdout_sha256': hashlib.sha256(child.stdout).hexdigest(),
    'raw_ssh_stderr_printed': False, 'sha256sums_sha256': sha(OUT / 'SHA256SUMS.txt')}
if child.returncode == 0:
    receipt['cloud_receipt'] = json.loads(child.stdout)
with (OUT / 'final-cloud-metadata-sync-receipt.json').open('x', encoding='utf-8', newline='\n') as stream:
    json.dump(receipt, stream, ensure_ascii=False, indent=2)
    stream.write('\n')
assert child.returncode == 0, 'Metadata sync failed; raw SSH stderr withheld.'
for path in files:
    if path.name != 'SHA256SUMS.txt':
        assert path.is_file()
print(json.dumps({'status': 'sealed_and_metadata_synced', 'metadata_files': len(metadata),
    'new_cloud_metadata_files': receipt['cloud_receipt']['new_files'],
    'sha256sums_sha256': receipt['sha256sums_sha256'],
    'sync_receipt_sha256': sha(OUT / 'final-cloud-metadata-sync-receipt.json')}), flush=True)
