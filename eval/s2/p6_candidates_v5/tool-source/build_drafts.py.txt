def rect(a, b, c, d):
    """Declare a rectangle explicitly. / 显式声明矩形。"""
    return [[a, b], [c, b], [c, d], [a, d]]


def domain(core, outer):
    """Keep core and possible extent separate. / 分开保存内域与可能范围。"""
    return {'core': core, 'outer': outer}


def from_crop(box, points):
    """Map hand-marked crop coordinates to the original frame. / 将裁剪图上的手工草稿坐标换算到原图。"""
    a, b, c, d = box
    return [[round(a + x * (c - a), 6), round(b + y * (d - b), 6)] for x, y in points]


def entity(ident, description, core, outer, bindings, notes):
    """Record a proposed physical group, not a verified unique defect.

记录拟定的实体分组，不将其记为已核实的独立缺陷。
"""
    return {'entity_id': ident, 'description': description, 'object_envelope': domain(core, outer),
            'candidate_refs': bindings, 'ai_notes': notes, 'grouping_confirmed_by_human': False,
            'human_verdict': None, 'physical_scale_known': False}


def refs(case, *pairs):
    """Bind every raw box to its original view and ordinal. / 将原始框绑定到原视图与序号。"""
    return [f'{case}:{view}:{ordinal}' for view, ordinal in pairs]
